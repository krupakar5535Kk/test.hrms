### 2.3 Complete Freeze Guard Trigger Definitions (All Mutation Paths)
Every freeze guard is implemented as a `BEFORE INSERT OR UPDATE OR DELETE` trigger on each respective table.
- **`INSERT`:** Asserts that the new record does not fall within a period where `is_closed = TRUE` or `status = 'FROZEN'`.
- **`DELETE`:** Asserts that the existing record does not belong to a closed or frozen period.
- **`UPDATE`:** **Must inspect BOTH `OLD` and `NEW` records.** If the row previously belonged to a frozen period, editing it is prohibited. If the modification attempts to change the date/period such that it enters a frozen period, it is prohibited.

```sql
-- ============================================================================
-- MASTER HELPER: Assert Period Is Open For Modification
-- ============================================================================
CREATE OR REPLACE FUNCTION public.fn_assert_period_is_open(
  p_organization_id UUID,
  p_date DATE,
  p_table_name TEXT,
  p_operation TEXT
) RETURNS VOID
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = pg_catalog, public
AS $$
DECLARE
  v_period_id UUID;
  v_is_closed BOOLEAN;
  v_run_status TEXT;
BEGIN
  IF p_date IS NULL OR p_organization_id IS NULL THEN
    RETURN;
  END IF;

  -- Lookup period covering target date
  SELECT pp.id, pp.is_closed, pr.status
  INTO v_period_id, v_is_closed, v_run_status
  FROM public.payroll_periods pp
  LEFT JOIN public.payroll_runs pr ON pr.payroll_period_id = pp.id 
    AND pr.status IN ('SUBMITTED_FOR_APPROVAL', 'APPROVED', 'PROCESSED')
  WHERE pp.organization_id = p_organization_id
    AND p_date BETWEEN pp.start_date AND pp.end_date
  LIMIT 1;

  IF FOUND THEN
    IF v_is_closed = TRUE THEN
      RAISE EXCEPTION 'Payroll freeze violation: Target date % falls in PERMANENTLY CLOSED period % (table %, op %)',
        p_date, v_period_id, p_table_name, p_operation
        USING ERRCODE = '55000'; -- object_not_in_prerequisite_state
    END IF;

    IF v_run_status IN ('SUBMITTED_FOR_APPROVAL', 'APPROVED', 'PROCESSED') THEN
      RAISE EXCEPTION 'Payroll freeze violation: Target date % falls in FROZEN period % with active run status % (table %, op %)',
        p_date, v_period_id, v_run_status, p_table_name, p_operation
        USING ERRCODE = '55000';
    END IF;
  END IF;
END;
$$;

-- ============================================================================
-- 1. attendance_days Freeze Guard
-- ============================================================================
CREATE OR REPLACE FUNCTION public.trg_freeze_guard_attendance_days()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = pg_catalog, public
AS $$
BEGIN
  IF TG_OP = 'INSERT' THEN
    PERFORM public.fn_assert_period_is_open(NEW.organization_id, NEW.attendance_date, 'attendance_days', TG_OP);
    RETURN NEW;
  ELSIF TG_OP = 'UPDATE' THEN
    PERFORM public.fn_assert_period_is_open(OLD.organization_id, OLD.attendance_date, 'attendance_days', TG_OP);
    IF (NEW.attendance_date <> OLD.attendance_date) OR (NEW.organization_id <> OLD.organization_id) THEN
      PERFORM public.fn_assert_period_is_open(NEW.organization_id, NEW.attendance_date, 'attendance_days', TG_OP);
    END IF;
    RETURN NEW;
  ELSIF TG_OP = 'DELETE' THEN
    PERFORM public.fn_assert_period_is_open(OLD.organization_id, OLD.attendance_date, 'attendance_days', TG_OP);
    RETURN OLD;
  END IF;
  RETURN NULL;
END;
$$;

CREATE TRIGGER trg_freeze_attendance_days
BEFORE INSERT OR UPDATE OR DELETE ON public.attendance_days
FOR EACH ROW EXECUTE FUNCTION public.trg_freeze_guard_attendance_days();

-- ============================================================================
-- 4. leave_applications Freeze Guard (Date Range Overlap)
-- ============================================================================
CREATE OR REPLACE FUNCTION public.trg_freeze_guard_leave_applications()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = pg_catalog, public
AS $$
DECLARE
  v_emp_org UUID;
  v_conflict RECORD;
BEGIN
  IF TG_OP IN ('UPDATE', 'DELETE') THEN
    SELECT organization_id INTO v_emp_org FROM public.employees WHERE id = OLD.employee_id;
    SELECT pp.id, pp.year, pp.month INTO v_conflict
    FROM public.payroll_periods pp
    LEFT JOIN public.payroll_runs pr ON pr.payroll_period_id = pp.id 
      AND pr.status IN ('SUBMITTED_FOR_APPROVAL', 'APPROVED', 'PROCESSED')
    WHERE pp.organization_id = v_emp_org
      AND (pp.is_closed = TRUE OR pr.id IS NOT NULL)
      AND daterange(pp.start_date, pp.end_date, '[]') && daterange(OLD.start_date, OLD.end_date, '[]')
    LIMIT 1;

    IF FOUND THEN
      RAISE EXCEPTION 'Payroll freeze violation: Leave mutation touches frozen period % (%-%)',
        v_conflict.id, v_conflict.year, v_conflict.month
        USING ERRCODE = '55000';
    END IF;
  END IF;

  IF TG_OP IN ('INSERT', 'UPDATE') THEN
    SELECT organization_id INTO v_emp_org FROM public.employees WHERE id = NEW.employee_id;
    SELECT pp.id, pp.year, pp.month INTO v_conflict
    FROM public.payroll_periods pp
    LEFT JOIN public.payroll_runs pr ON pr.payroll_period_id = pp.id 
      AND pr.status IN ('SUBMITTED_FOR_APPROVAL', 'APPROVED', 'PROCESSED')
    WHERE pp.organization_id = v_emp_org
      AND (pp.is_closed = TRUE OR pr.id IS NOT NULL)
      AND daterange(pp.start_date, pp.end_date, '[]') && daterange(NEW.start_date, NEW.end_date, '[]')
    LIMIT 1;

    IF FOUND THEN
      RAISE EXCEPTION 'Payroll freeze violation: Proposed leave touches frozen period % (%-%)',
        v_conflict.id, v_conflict.year, v_conflict.month
        USING ERRCODE = '55000';
    END IF;
    RETURN NEW;
  END IF;

  RETURN OLD;
END;
$$;
```

### 2.4 Deterministic Versioned Snapshot & Staleness Protocol
To eliminate previous flawed references to uncanonicalized JSONB serialization, the payroll snapshot is governed by an explicit versioned protocol:

1. **Schema Versioning:** The payload specifies an explicit top-level attribute `"snapshot_version": "1.0"`.
2. **Alphabetical Attribute Ordering:** Object keys are ordered strictly alphabetically at every level.
3. **Fixed Numeric Formatting:** All currency amounts are represented as strict integer paise (`BIGINT`). Minute quantities are formatted as integer minutes. Minute rates are rounded to exactly 6 decimal places: `ROUND(rate::numeric, 6)::text`.
4. **Deterministic Sorting:** Line items are ordered by `employee_id ASC`.
5. **Cryptographic Staleness Token:** The snapshot text is encoded to UTF-8 bytes and digested via `public.digest(utf8_bytes, 'sha256')`.

```sql
CREATE OR REPLACE FUNCTION public.fn_generate_deterministic_payroll_snapshot(
  p_payroll_run_id UUID
) RETURNS JSONB
LANGUAGE plpgsql
STABLE
SECURITY DEFINER
SET search_path = pg_catalog, public
AS $$
DECLARE
  v_run RECORD;
  v_line_items JSONB;
  v_canonical_text TEXT;
  v_sha256_hash TEXT;
  v_final_snapshot JSONB;
BEGIN
  -- 1. Fetch run header
  SELECT pr.id, pr.payroll_period_id, pp.organization_id, pp.year, pp.month,
         pr.total_gross_paise, pr.total_deductions_paise, pr.total_net_paise
  INTO v_run
  FROM public.payroll_runs pr
  JOIN public.payroll_periods pp ON pp.id = pr.payroll_period_id
  WHERE pr.id = p_payroll_run_id;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Payroll run % not found', p_payroll_run_id USING ERRCODE = 'P0002';
  END IF;

  -- 2. Aggregate deterministic, sorted array of payroll line items
  WITH sorted_items AS (
    SELECT
      pi.employee_id::text AS employee_id,
      pi.monthly_salary_reference_paise AS monthly_ref_paise,
      pi.standard_scheduled_minutes AS scheduled_mins,
      ROUND(pi.minute_rate_paise::numeric, 6)::text AS minute_rate_paise,
      pi.approved_payable_minutes AS payable_mins,
      pi.actual_worked_minutes AS worked_mins,
      pi.unpaid_leave_minutes AS unpaid_mins,
      pi.absent_minutes AS absent_mins,
      pi.earned_salary_paise AS earned_paise,
      pi.total_allowances_paise AS allowances_paise,
      pi.total_adjustments_paise AS adjustments_paise,
      pi.total_deductions_paise AS deductions_paise,
      pi.net_salary_paise AS net_paise
    FROM public.payroll_items pi
    WHERE pi.payroll_run_id = p_payroll_run_id
    ORDER BY pi.employee_id ASC
  )
  SELECT jsonb_agg(
    jsonb_build_object(
      'absent_mins', absent_mins,
      'adjustments_paise', adjustments_paise,
      'allowances_paise', allowances_paise,
      'deductions_paise', deductions_paise,
      'earned_paise', earned_paise,
      'employee_id', employee_id,
      'minute_rate_paise', minute_rate_paise,
      'monthly_ref_paise', monthly_ref_paise,
      'net_paise', net_paise,
      'payable_mins', payable_mins,
      'scheduled_mins', scheduled_mins,
      'unpaid_mins', unpaid_mins,
      'worked_mins', worked_mins
    ) ORDER BY employee_id ASC
  ) INTO v_line_items
  FROM sorted_items;

  -- 3. Construct canonical text representation
  v_canonical_text := jsonb_build_object(
    'header', jsonb_build_object(
      'month', v_run.month,
      'org_id', v_run.organization_id::text,
      'period_id', v_run.payroll_period_id::text,
      'run_id', v_run.id::text,
      'total_deductions_paise', v_run.total_deductions_paise,
      'total_gross_paise', v_run.total_gross_paise,
      'total_net_paise', v_run.total_net_paise,
      'year', v_run.year
    ),
    'items', COALESCE(v_line_items, '[]'::jsonb),
    'snapshot_version', '1.0'
  )::text;

  -- 4. Calculate SHA-256 staleness digest
  v_sha256_hash := encode(public.digest(convert_to(v_canonical_text, 'UTF8'), 'sha256'), 'hex');

  -- 5. Return complete versioned snapshot
  RETURN jsonb_build_object(
    'canonical_payload', v_canonical_text::jsonb,
    'sha256_hash', v_sha256_hash,
    'generated_at', to_char(now() AT TIME ZONE 'UTC', 'YYYY-MM-DD"T"HH24:MI:SS.US"Z"')
  );
END;
$$;
```

### 2.5 Approval-Time Validation & Recompilation Rules
When a Payroll Approver submits an approval request (`POST /api/v1/payroll/runs/:id/approve`):

1. **Re-Verification Step:** The stored procedure `approve_payroll_run(p_run_id UUID)` executes inside an explicit transaction:
   ```sql
   SELECT id FROM public.payroll_periods WHERE id = v_period_id FOR UPDATE;
   ```
2. **Hash Comparison:** It computes `v_current_hash := fn_generate_deterministic_payroll_snapshot(p_run_id)->>'sha256_hash'`.
3. **Staleness Assertion:**
   ```sql
   IF v_current_hash <> v_stored_run_hash THEN
     RAISE EXCEPTION 'PAYROLL_STALE_SNAPSHOT: Source attendance or compensation data was modified since calculation. Recompilation required.'
       USING ERRCODE = 'P0004';
   END IF;
   ```
4. **Recompilation Rule:** If `PAYROLL_STALE_SNAPSHOT` is raised:
   - Approval is aborted.
   - The status remains `DRAFT`.
   - The user must trigger `recompile_payroll_run(p_run_id)`, which re-scans attendance days, re-derives line items, updates the snapshot hash, and requires fresh approval.

### 3.3 Fail-Closed Tenant & Actor Identity Validation Trigger
The trigger below rejects any mutation where `app.current_org_id` or `app.current_user_id` is null, empty, or mismatched with the target tuple:

```sql
CREATE OR REPLACE FUNCTION public.trg_assert_tenant_and_actor_integrity()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = pg_catalog, public
AS $$
DECLARE
  v_session_org_id TEXT;
  v_session_user_id TEXT;
BEGIN
  -- Read transactional session GUCs
  v_session_org_id  := current_setting('app.current_org_id', true);
  v_session_user_id := current_setting('app.current_user_id', true);

  -- FAIL-CLOSED ASSERTIONS
  IF v_session_org_id IS NULL OR v_session_org_id = '' THEN
    RAISE EXCEPTION 'Access denied: Session variable app.current_org_id is unset (SQLSTATE 42501)'
      USING ERRCODE = 'insufficient_privilege';
  END IF;

  IF v_session_user_id IS NULL OR v_session_user_id = '' THEN
    RAISE EXCEPTION 'Access denied: Session variable app.current_user_id is unset (SQLSTATE 42501)'
      USING ERRCODE = 'insufficient_privilege';
  END IF;

  -- TENANT VALIDATION
  IF TG_OP IN ('INSERT', 'UPDATE') THEN
    IF TG_TABLE_NAME IN ('offices', 'departments', 'users', 'employees', 'attendance_days', 
                         'leave_types', 'holidays', 'payroll_periods', 'audit_logs') THEN
      IF NEW.organization_id::text <> v_session_org_id THEN
        RAISE EXCEPTION 'Cross-tenant violation: Target organization % does not match session % (SQLSTATE 42501)',
          NEW.organization_id, v_session_org_id
          USING ERRCODE = 'insufficient_privilege';
      END IF;
    END IF;
  END IF;

  RETURN COALESCE(NEW, OLD);
END;
$$;
```

### 3.4 Application-Controlled GUCs vs Independently Authenticated Identity
To preserve architectural integrity, the boundary between application-controlled GUCs and independently authenticated identity is formalized:

```
[Untrusted Client Request]
       │
       ▼
[Express Authentication Gateway]
   1. Validates Supabase JWT (RS256 signature, expiry, audience = 'authenticated')
   2. Extracts auth_user_id (claims.sub)
   3. Resolves user from DB: resolve_user_auth_context(claims.sub) -> (user_id, organization_id, role, is_active)
   4. Acquires pooled client: client = await pool.connect();
   5. Starts transaction: BEGIN;
   6. Sets local GUCs:
        SELECT set_config('app.current_user_id', user_id, true),
               set_config('app.current_org_id', organization_id, true),
               set_config('app.current_user_role', role, true);
       │
       ▼
[PostgreSQL Enforcement Engine]
   1. RLS Policies assert: organization_id = current_org_id()
   2. Triggers assert: trg_assert_tenant_and_actor_integrity()
   3. hrms_app_user lacks BYPASSRLS, SUPERUSER, and ALTER SYSTEM.
   4. Local GUCs automatically expire on COMMIT / ROLLBACK.
```

**Boundary Distinction:** The database engine does not communicate directly with the mobile client or Supabase GoTrue. It trusts **only** the transactional local GUCs set by the authenticated Express broker. If an attacker attempts to inject GUCs via client headers, the Express gateway strips untrusted headers and sets GUCs using exclusively verified JWT claims.

## 4. SECTION C: LEGACY EMPLOYEE IMPORT SECURITY, IDENTITY & ATOMICITY

### 4.1 Remediation of the NULL Role Fail-Open Vulnerability
In previous drafts (A3.10 / A3.11), the role check was written as:
```sql
v_caller_role := current_setting('app.current_user_role', true);
IF v_caller_role NOT IN ('MAINTENANCE_ADMIN', 'SUPER_ADMIN') THEN ...
```
In SQL three-valued logic, `NULL NOT IN ('A', 'B')` evaluates to `NULL` (falsy in boolean conditional expressions). Consequently, if `app.current_user_role` was unset (NULL), the exception branch was **skipped**, allowing unprivileged callers to execute administrative imports.

**The Fail-Closed Remediation:**
```sql
IF v_caller_role IS NULL OR v_caller_role NOT IN ('MAINTENANCE_ADMIN', 'SUPER_ADMIN') THEN
  RAISE EXCEPTION 'Unauthorized: Caller role % cannot perform legacy employee import (SQLSTATE 42501)',
    COALESCE(v_caller_role, 'NULL')
    USING ERRCODE = 'insufficient_privilege';
END IF;
```

## 5. SECTION D: DATABASE LOCKING HIERARCHY, ENGINE EXECUTION & CONCURRENCY

### 5.1 The Engine Pre-Trigger Row-Lock Conflict Resolved
In PostgreSQL, the query:
```sql
UPDATE attendance_days SET net_payable_minutes = 480 WHERE id = $1;
```
executes through the following physical steps:
1. Index scan locates the heap tuple on disk.
2. **The engine immediately acquires an exclusive row lock (`ExclusiveLock` / `FOR UPDATE`) on the `attendance_days` tuple.**
3. The engine fires the `BEFORE UPDATE` trigger (`trg_freeze_guard_attendance_days`).
4. If that trigger executes `SELECT 1 FROM payroll_periods WHERE ... FOR SHARE;`, the lock acquisition sequence is:
   $$\text{Tuple Locked First: } L_2 (\text{attendance\_days}) \longrightarrow \text{Parent Lock Attempted: } L_1 (\text{payroll\_periods})$$

**Why This Creates Inversion Deadlocks:**
A concurrent payroll calculation or freeze transaction follows the natural hierarchical order:
1. Locks $L_1$ (`payroll_periods`) `FOR UPDATE`.
2. Attempts to scan/lock child records $L_2$ (`attendance_days`) `FOR UPDATE`.

If Transaction 1 holds $L_2$ and waits for $L_1$, while Transaction 2 holds $L_1$ and waits for $L_2$, **PostgreSQL detects a deadlock cycle and aborts one transaction (`SQLSTATE 40P01`)**.

**The Architectural Resolution:**
1. **Triggers Never Lock:** All `BEFORE` triggers act as **read-only validation guards**. They execute pure `SELECT` queries without `FOR SHARE` or `FOR UPDATE`.
2. **Top-Down Lock Acquisition at Transaction Entry:** All write paths (Express routes, Celery background tasks, stored procedures) **MUST acquire the parent lock ($L_1$) prior to executing child DML.**
3. **Hierarchy:**
   $$L_1 (\text{payroll\_periods}) \prec L_2 (\text{attendance\_days}) \prec L_3 (\text{attendance\_events / corrections}) \prec L_4 (\text{audit\_logs})$$

---

### 5.2 Reconciling Direct DML Privileges with the Locking Protocol
If the application role `hrms_app_user` possesses direct `UPDATE` privileges on `attendance_days`, an ad-hoc query could bypass the top-down locking requirement and cause lock inversions.

**Architectural Controls:**
1. **Primary Defense (Transaction Middleware):** The Express data access layer enforces top-down locking via a mandatory transaction helper (`executeAttendanceMutation()`) that executes:
   ```sql
   SELECT id FROM public.payroll_periods WHERE id = $period_id FOR SHARE;
   ```
   prior to issuing any `UPDATE attendance_days`.
2. **Secondary Defense (Bounded Timeouts):** All application transactions configure:
   - `SET LOCAL lock_timeout = '3s';`
   - `SET LOCAL statement_timeout = '5s';`
   If an unexpected lock conflict occurs, the transaction aborts cleanly within 3 seconds, releasing all locks without hanging the connection pool.
3. **Deadlock Recovery:** The Express gateway implements an automatic retry mechanism with exponential backoff and jitter (maximum 3 attempts) for `SQLSTATE 40P01` (deadlock detected) and `SQLSTATE 55P03` (lock not available).

---

### 5.3 Realistic Deadlock Risks & Concurrency Matrix
In compliance with strict audit standards, deadlocks cannot be claimed as "zero" without empirical testing. The table below identifies remaining deadlock vectors under high concurrency:

| Concurrency Scenario | Competing Transactions | Competing Resources | Deadlock Risk Level | Mitigation & Recovery | Status |
| :--- | :--- | :--- | :--- | :--- | :--- |
| **C01: Punch Sync vs Period Freeze** | Tx1: Offline biometric batch sync<br>Tx2: Admin freeze run | `payroll_periods` ($L_1$), `attendance_days` ($L_2$) | **LOW** | Tx1 acquires $L_1$ `FOR SHARE`; Tx2 blocks on $L_1$ `FOR UPDATE`. Clean serialization. | `DESIGN-LEVEL, NOT TESTED` |
| **C02: Multi-Day Batch Punch Sync** | Tx1: Batch sync for Days [D1, D2]<br>Tx2: Batch sync for Days [D2, D1] | `attendance_days` (D1, D2) | **HIGH (if unsorted)** | Application layer sorts days in strict ascending date order: `ORDER BY attendance_date ASC`. | `DESIGN-LEVEL, NOT TESTED` |
| **C03: Manager Day Edit vs Punch Sync** | Tx1: Manager edits Day D<br>Tx2: Worker check-out on Day D | `attendance_days` (D) | **MEDIUM** | Both compete for row lock on D. First committer wins; second committer re-reads committed version. | `DESIGN-LEVEL, NOT TESTED` |
| **C04: Dual Payroll Calculations** | Tx1: Worker A calculates period P<br>Tx2: Worker B calculates period P | `payroll_periods` (P) | **VERY LOW** | `SELECT id FROM payroll_periods WHERE id = P FOR UPDATE` serializes workers immediately. | `DESIGN-LEVEL, NOT TESTED` |
| **C05: Leave Approval vs Day Correction** | Tx1: HR approves leave for Day D<br>Tx2: Admin corrects punch on Day D | `attendance_days` (D) | **MEDIUM** | Row lock on `attendance_days`. Serialized execution via standard MVCC row locking. | `DESIGN-LEVEL, NOT TESTED` |

## 6. SECTION E: AUTHENTICATION, RLS BOOTSTRAP & EXPRESS LIFECYCLE

### 6.1 Corrected `SECURITY DEFINER` and `FORCE RLS` Bootstrap Design
Under PostgreSQL's RLS engine:
- If a table has `ENABLE ROW LEVEL SECURITY` (without `FORCE RLS`), the **table owner** (`hrms_schema_owner`) and superusers bypass RLS policies.
- If a table has `FORCE ROW LEVEL SECURITY`, RLS applies to **all roles**, including the table owner.

**The Bootstrap Deadlock Problem:**
When a user attempts to log in, the backend must query `users` to look up their `organization_id`, `role`, and `is_active` status. However:
- The session GUC `app.current_org_id` cannot be set *before* querying the user's organization!
- If `users` had `FORCE RLS`, the query `SELECT * FROM users WHERE auth_user_id = $1` would evaluate the policy `organization_id = current_org_id()`.
- Because `current_org_id()` is NULL, the policy evaluates to `FALSE`, returning 0 rows and bricking authentication!

**The Architectural Solution:**
1. `organizations` and `users`: Configured with **`ENABLE ROW LEVEL SECURITY`** (WITHOUT `FORCE RLS`).
2. Operational tables (`employees`, `attendance_days`, `payroll_runs`, etc.): Configured with **`FORCE ROW LEVEL SECURITY`**.
3. Bootstrap lookup procedure `resolve_user_auth_context()`: Owned by `hrms_schema_owner` with `SECURITY DEFINER` and `SET search_path = pg_catalog, public`.
4. When `hrms_app_user` calls `resolve_user_auth_context()`, it executes with table owner privileges, bypassing RLS on `users` and safely returning the tenant context.

```sql
CREATE OR REPLACE FUNCTION public.resolve_user_auth_context(
  p_auth_user_id UUID
) RETURNS TABLE (
  user_id UUID,
  organization_id UUID,
  user_role TEXT,
  is_active BOOLEAN,
  org_active BOOLEAN
)
LANGUAGE plpgsql
STABLE
SECURITY DEFINER
SET search_path = pg_catalog, public
AS $$
BEGIN
  RETURN QUERY
  SELECT
    u.id AS user_id,
    u.organization_id,
    ur.role::text AS user_role,
    u.is_active,
    o.is_active AS org_active
  FROM public.users u
  JOIN public.organizations o ON o.id = u.organization_id
  LEFT JOIN public.user_roles ur ON ur.user_id = u.id
  WHERE u.id = p_auth_user_id OR u.phone_number = (
    -- Supports lookup by phone or ID
    SELECT phone_number FROM public.users WHERE id = p_auth_user_id
  )
  LIMIT 1;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Authentication context resolution failed: User % not found', p_auth_user_id
      USING ERRCODE = 'P0002';
  END IF;
END;
$$;

ALTER FUNCTION public.resolve_user_auth_context(UUID) OWNER TO hrms_schema_owner;
GRANT EXECUTE ON FUNCTION public.resolve_user_auth_context(UUID) TO hrms_app_user;
```

### 6.2 Express Transactional Client Management & Error Handling
To eliminate connection pooling contamination, a dedicated transactional middleware manages client checkouts, GUC configuration, and defensive cleanup:

```typescript
import { Pool, PoolClient } from 'pg';
import { Request, Response, NextFunction } from 'express';

const dbPool = new Pool({
  connectionString: process.env.DATABASE_URL,
  max: 50,
  idleTimeoutMillis: 30000,
});

export async function withTenantTransaction<T>(
  userId: string,
  orgId: string,
  userRole: string,
  callback: (client: PoolClient) => Promise<T>
): Promise<T> {
  const client = await dbPool.connect();
  try {
    // 1. Begin atomic transaction
    await client.query('BEGIN');

    // 2. Set transaction-scoped local GUCs (3rd argument `true` scopes strictly to transaction)
    await client.query(
      `SELECT
         set_config('app.current_user_id', $1, true),
         set_config('app.current_org_id', $2, true),
         set_config('app.current_user_role', $3, true)`,
      [userId, orgId, userRole]
    );

    // 3. Execute protected business logic over the EXACT SAME client
    const result = await callback(client);

    // 4. Commit transaction
    await client.query('COMMIT');
    return result;
  } catch (error) {
    // 5. Rollback transaction on failure (never suppress error)
    try {
      await client.query('ROLLBACK');
    } catch (rollbackErr) {
      // If rollback fails, destroy the socket completely
      client.release(rollbackErr);
      throw error;
    }
    throw error;
  } finally {
    // 6. Defensive connection sanitization before returning to pool
    try {
      await client.query('DISCARD SEQUENCES; RESET ALL;');
      client.release();
    } catch (cleanupErr) {
      // CRITICAL: If cleanup fails, destroy client to prevent contaminated socket reuse
      client.release(cleanupErr);
    }
  }
}
```
