```sql
CREATE OR REPLACE FUNCTION public.fn_generate_source_sensitive_payroll_snapshot(
  p_payroll_run_id UUID
) RETURNS JSONB
LANGUAGE plpgsql
STABLE
SECURITY DEFINER
SET search_path = pg_catalog, public
AS $$
DECLARE
  v_run RECORD;
  v_period RECORD;
  v_source_inputs JSONB;
  v_calculated_outputs JSONB;
  v_canonical_text TEXT;
  v_sha256_hash TEXT;
BEGIN
  -- 1. Fetch payroll run header and period details
  SELECT pr.id, pr.payroll_period_id, pr.status, pr.prepared_by,
         pr.total_gross_paise, pr.total_deductions_paise, pr.total_net_paise
  INTO v_run
  FROM public.payroll_runs pr
  WHERE pr.id = p_payroll_run_id;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Payroll run % not found', p_payroll_run_id USING ERRCODE = 'P0002';
  END IF;

  SELECT pp.id, pp.organization_id, pp.year, pp.month, pp.start_date, pp.end_date,
         pp.scheduled_working_days, pp.standard_monthly_minutes, pp.is_closed
  INTO v_period
  FROM public.payroll_periods pp
  WHERE pp.id = v_run.payroll_period_id;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Payroll period % not found for run %', v_run.payroll_period_id, p_payroll_run_id USING ERRCODE = 'P0002';
  END IF;

  -- 2. Aggregate deterministic, sorted SOURCE INPUT TELEMETRY
  WITH emp_attendance AS (
    SELECT
      ad.employee_id,
      SUM(ad.gross_elapsed_minutes) AS total_gross_elapsed_mins,
      SUM(ad.lunch_deducted_minutes) AS total_lunch_deducted_mins,
      SUM(ad.approved_post18_minutes) AS total_approved_post18_mins,
      SUM(ad.net_payable_minutes) AS total_net_payable_mins,
      COUNT(*) FILTER (WHERE ad.day_status = 'PRESENT') AS days_present,
      COUNT(*) FILTER (WHERE ad.day_status = 'ABSENT') AS days_absent,
      COUNT(*) FILTER (WHERE ad.day_status = 'ON_LEAVE') AS days_on_leave
    FROM public.attendance_days ad
    WHERE ad.organization_id = v_period.organization_id
      AND ad.attendance_date BETWEEN v_period.start_date AND v_period.end_date
    GROUP BY ad.employee_id
  ),
  emp_leaves AS (
    SELECT
      la.employee_id,
      COALESCE(SUM(la.total_days) FILTER (WHERE lt.is_paid = TRUE), 0.0) AS paid_leave_days,
      COALESCE(SUM(la.total_days) FILTER (WHERE lt.is_paid = FALSE), 0.0) AS unpaid_leave_days
    FROM public.leave_applications la
    JOIN public.leave_types lt ON lt.id = la.leave_type_id
    WHERE la.status = 'APPROVED'
      AND daterange(v_period.start_date, v_period.end_date, '[]') && daterange(la.start_date, la.end_date, '[]')
    GROUP BY la.employee_id
  ),
  emp_salaries AS (
    SELECT DISTINCT ON (es.employee_id)
      es.employee_id,
      es.monthly_salary_reference_paise,
      es.effective_from,
      es.effective_to
    FROM public.employee_salaries es
    WHERE es.effective_from <= v_period.end_date
      AND (es.effective_to IS NULL OR es.effective_to >= v_period.start_date)
    ORDER BY es.employee_id, es.effective_from DESC
  ),
  emp_adjustments AS (
    SELECT
      epa.employee_id,
      COALESCE(SUM(epa.amount_paise), 0) AS total_adjustments_paise
    FROM public.employee_payroll_adjustments epa
    WHERE epa.payroll_period_id = v_period.id
    GROUP BY epa.employee_id
  ),
  combined_inputs AS (
    SELECT
      e.id::text AS employee_id,
      e.employee_code,
      e.employment_status::text AS employment_status,
      e.joining_date::text AS joining_date,
      COALESCE(e.exit_date::text, '') AS exit_date,
      COALESCE(es.monthly_salary_reference_paise, 0)::text AS salary_reference_paise,
      COALESCE(ea.total_gross_elapsed_mins, 0)::text AS gross_elapsed_mins,
      COALESCE(ea.total_lunch_deducted_mins, 0)::text AS lunch_deducted_mins,
      COALESCE(ea.total_approved_post18_mins, 0)::text AS approved_post18_mins,
      COALESCE(ea.total_net_payable_mins, 0)::text AS net_payable_mins,
      COALESCE(el.paid_leave_days, 0.0)::text AS paid_leave_days,
      COALESCE(el.unpaid_leave_days, 0.0)::text AS unpaid_leave_days,
      COALESCE(adj.total_adjustments_paise, 0)::text AS adjustments_paise
    FROM public.employees e
    LEFT JOIN emp_salaries es ON es.employee_id = e.id
    LEFT JOIN emp_attendance ea ON ea.employee_id = e.id
    LEFT JOIN emp_leaves el ON el.employee_id = e.id
    LEFT JOIN emp_adjustments adj ON adj.employee_id = e.id
    WHERE e.organization_id = v_period.organization_id
      AND e.joining_date <= v_period.end_date
      AND (e.exit_date IS NULL OR e.exit_date >= v_period.start_date)
    ORDER BY e.id ASC
  )
  SELECT jsonb_agg(
    jsonb_build_object(
      'adjustments_paise', adjustments_paise,
      'approved_post18_mins', approved_post18_mins,
      'employee_code', employee_code,
      'employee_id', employee_id,
      'employment_status', employment_status,
      'exit_date', exit_date,
      'gross_elapsed_mins', gross_elapsed_mins,
      'joining_date', joining_date,
      'lunch_deducted_mins', lunch_deducted_mins,
      'net_payable_mins', net_payable_mins,
      'paid_leave_days', paid_leave_days,
      'salary_reference_paise', salary_reference_paise,
      'unpaid_leave_days', unpaid_leave_days
    ) ORDER BY employee_id ASC
  ) INTO v_source_inputs
  FROM combined_inputs;

  -- 3. Aggregate deterministic, sorted DERIVED OUTPUT LINE ITEMS
  WITH sorted_outputs AS (
    SELECT
      pi.employee_id::text AS employee_id,
      pi.monthly_salary_reference_paise::text AS monthly_ref_paise,
      pi.standard_scheduled_minutes::text AS scheduled_mins,
      ROUND(pi.minute_rate_paise::numeric, 6)::text AS minute_rate_paise,
      pi.approved_payable_minutes::text AS payable_mins,
      pi.actual_worked_minutes::text AS worked_mins,
      pi.unpaid_leave_minutes::text AS unpaid_mins,
      pi.absent_minutes::text AS absent_mins,
      pi.earned_salary_paise::text AS earned_paise,
      pi.total_allowances_paise::text AS allowances_paise,
      pi.total_adjustments_paise::text AS adjustments_paise,
      pi.total_deductions_paise::text AS deductions_paise,
      pi.net_salary_paise::text AS net_paise
    FROM public.payroll_items pi
    WHERE pi.payroll_run_id = p_payroll_run_id
    ORDER BY pi.employee_id ASC
  )
  SELECT jsonb_agg(
    jsonb_build_object(
      'absent_mins', absent_mins,
      'adjustments_paise', adjustments_paise,
      'allowances_paise', allowances_paise,
      'deductions_paise', deductions_paise,
      'earned_paise', earned_paise,
      'employee_id', employee_id,
      'minute_rate_paise', minute_rate_paise,
      'monthly_ref_paise', monthly_ref_paise,
      'net_paise', net_paise,
      'payable_mins', payable_mins,
      'scheduled_mins', scheduled_mins,
      'unpaid_mins', unpaid_mins,
      'worked_mins', worked_mins
    ) ORDER BY employee_id ASC
  ) INTO v_calculated_outputs
  FROM sorted_outputs;

  -- 4. Construct canonical JSON string representation
  v_canonical_text := jsonb_build_object(
    'calculated_outputs', COALESCE(v_calculated_outputs, '[]'::jsonb),
    'period_header', jsonb_build_object(
      'end_date', v_period.end_date::text,
      'is_closed', v_period.is_closed,
      'month', v_period.month,
      'org_id', v_period.organization_id::text,
      'period_id', v_period.id::text,
      'run_id', v_run.id::text,
      'scheduled_working_days', v_period.scheduled_working_days,
      'standard_monthly_minutes', v_period.standard_monthly_minutes,
      'start_date', v_period.start_date::text,
      'total_deductions_paise', v_run.total_deductions_paise::text,
      'total_gross_paise', v_run.total_gross_paise::text,
      'total_net_paise', v_run.total_net_paise::text,
      'year', v_period.year
    ),
    'snapshot_version', '2.0',
    'source_inputs', COALESCE(v_source_inputs, '[]'::jsonb)
  )::text;

  -- 5. Calculate cryptographic SHA-256 digest
  v_sha256_hash := encode(public.digest(convert_to(v_canonical_text, 'UTF8'), 'sha256'), 'hex');

  RETURN jsonb_build_object(
    'canonical_payload', v_canonical_text::jsonb,
    'sha256_hash', v_sha256_hash,
    'generated_at', to_char(now() AT TIME ZONE 'UTC', 'YYYY-MM-DD"T"HH24:MI:SS.US"Z"')
  );
END;
$$;
```

### 2.2 Complete Calculation Engine Definition (`fn_execute_period_calculation_engine`)
- **Original Defect:** Identified in A3.13.1 §2.2. A3.13 line 553 invoked `fn_execute_period_calculation_engine` during recompilation without supplying its PL/pgSQL function definition or detailing its calculation rules.
- **Architectural Correction:** Provide the full calculation procedure strictly adhering to approved decisions:
  - Working-hour minute rate: `monthly_salary_reference_paise / standard_monthly_minutes` (stored as `numeric(14, 6)`).
  - DEC-01 statutory lunch deduction: already computed on `attendance_days` daily aggregate.
  - Zero unapproved additions: Paid leave is compensated at standard rate only if `leave_types.is_paid = TRUE`; statutory holidays only if `holidays.is_paid = TRUE`. No unapproved overtime rate multipliers or unvetted tax formulas are introduced.

```sql
CREATE OR REPLACE FUNCTION public.fn_execute_period_calculation_engine(
  p_payroll_period_id UUID,
  p_payroll_run_id UUID
) RETURNS VOID
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = pg_catalog, public
AS $$
DECLARE
  v_period RECORD;
  v_emp RECORD;
  v_minute_rate NUMERIC(14, 6);
  v_actual_worked_mins INT;
  v_approved_payable_mins INT;
  v_unpaid_leave_mins INT;
  v_absent_mins INT;
  v_earned_salary_paise BIGINT;
  v_total_adjustments_paise BIGINT;
  v_total_deductions_paise BIGINT;
  v_net_salary_paise BIGINT;
  v_run_gross_paise BIGINT := 0;
  v_run_deductions_paise BIGINT := 0;
  v_run_net_paise BIGINT := 0;
BEGIN
  -- 1. Fetch period calendar
  SELECT id, organization_id, start_date, end_date, standard_monthly_minutes, scheduled_working_days
  INTO v_period
  FROM public.payroll_periods
  WHERE id = p_payroll_period_id;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Payroll period % not found', p_payroll_period_id USING ERRCODE = 'P0002';
  END IF;

  -- 2. Clear existing draft items for this run
  DELETE FROM public.payroll_items WHERE payroll_run_id = p_payroll_run_id;

  -- 3. Iterate across all eligible employees
  FOR v_emp IN
    SELECT e.id AS employee_id,
           COALESCE(es.monthly_salary_reference_paise, 0) AS monthly_salary_reference_paise
    FROM public.employees e
    LEFT JOIN LATERAL (
      SELECT es_inner.monthly_salary_reference_paise
      FROM public.employee_salaries es_inner
      WHERE es_inner.employee_id = e.id
        AND es_inner.effective_from <= v_period.end_date
        AND (es_inner.effective_to IS NULL OR es_inner.effective_to >= v_period.start_date)
      ORDER BY es_inner.effective_from DESC
      LIMIT 1
    ) es ON true
    WHERE e.organization_id = v_period.organization_id
      AND e.employment_status IN ('CONFIRMED', 'PROBATION')
      AND e.joining_date <= v_period.end_date
      AND (e.exit_date IS NULL OR e.exit_date >= v_period.start_date)
  LOOP
    -- Calculate minute rate
    IF v_period.standard_monthly_minutes > 0 THEN
      v_minute_rate := ROUND((v_emp.monthly_salary_reference_paise::numeric / v_period.standard_monthly_minutes::numeric), 6);
    ELSE
      v_minute_rate := 0.000000;
    END IF;

    -- Aggregate attendance minutes for employee
    SELECT
      COALESCE(SUM(ad.gross_elapsed_minutes - ad.lunch_deducted_minutes), 0),
      COALESCE(SUM(ad.net_payable_minutes), 0),
      COALESCE(COUNT(*) FILTER (WHERE ad.day_status = 'ABSENT') * 480, 0)
    INTO v_actual_worked_mins, v_approved_payable_mins, v_absent_mins
    FROM public.attendance_days ad
    WHERE ad.employee_id = v_emp.employee_id
      AND ad.attendance_date BETWEEN v_period.start_date AND v_period.end_date;

    -- Aggregate unpaid leave minutes (assuming standard 8-hour / 480-minute day)
    SELECT COALESCE(SUM(la.total_days * 480), 0)::int
    INTO v_unpaid_leave_mins
    FROM public.leave_applications la
    JOIN public.leave_types lt ON lt.id = la.leave_type_id
    WHERE la.employee_id = v_emp.employee_id
      AND la.status = 'APPROVED'
      AND lt.is_paid = FALSE
      AND daterange(v_period.start_date, v_period.end_date, '[]') && daterange(la.start_date, la.end_date, '[]');

    -- Calculate base earned salary in paise
    v_earned_salary_paise := ROUND(v_approved_payable_mins::numeric * v_minute_rate);

    -- Fetch period adjustments (bonuses, advances, recoveries)
    SELECT
      COALESCE(SUM(amount_paise) FILTER (WHERE amount_paise > 0), 0),
      COALESCE(ABS(SUM(amount_paise) FILTER (WHERE amount_paise < 0)), 0)
    INTO v_total_adjustments_paise, v_total_deductions_paise
    FROM public.employee_payroll_adjustments
    WHERE payroll_period_id = p_payroll_period_id
      AND employee_id = v_emp.employee_id;

    -- Final net calculation
    v_net_salary_paise := v_earned_salary_paise + v_total_adjustments_paise - v_total_deductions_paise;
    IF v_net_salary_paise < 0 THEN
      v_net_salary_paise := 0;
    END IF;

    -- Insert calculated line item
    INSERT INTO public.payroll_items (
      payroll_run_id,
      employee_id,
      monthly_salary_reference_paise,
      standard_scheduled_minutes,
      minute_rate_paise,
      approved_payable_minutes,
      actual_worked_minutes,
      unpaid_leave_minutes,
      absent_minutes,
      earned_salary_paise,
      total_allowances_paise,
      total_adjustments_paise,
      total_deductions_paise,
      net_salary_paise,
      created_at
    ) VALUES (
      p_payroll_run_id,
      v_emp.employee_id,
      v_emp.monthly_salary_reference_paise,
      v_period.standard_monthly_minutes,
      v_minute_rate,
      v_approved_payable_mins,
      v_actual_worked_mins,
      v_unpaid_leave_mins,
      v_absent_mins,
      v_earned_salary_paise,
      0, -- Allowances baseline
      v_total_adjustments_paise,
      v_total_deductions_paise,
      v_net_salary_paise,
      now()
    );

    v_run_gross_paise := v_run_gross_paise + v_earned_salary_paise + v_total_adjustments_paise;
    v_run_deductions_paise := v_run_deductions_paise + v_total_deductions_paise;
    v_run_net_paise := v_run_net_paise + v_net_salary_paise;
  END LOOP;

  -- Update payroll run totals
  UPDATE public.payroll_runs
  SET total_gross_paise = v_run_gross_paise,
      total_deductions_paise = v_run_deductions_paise,
      total_net_paise = v_run_net_paise,
      updated_at = now()
  WHERE id = p_payroll_run_id;
END;
$$;
```

### 2.3 Deterministic Canonical Serialization & RFC 8785 Clarification
- **Original Defect:** Prior documents claimed compliance with "RFC 8785 (JSON Canonicalization Scheme)" on raw PostgreSQL `jsonb`.
- **Architectural Correction:** RFC 8785 defines canonical byte serialization rules (e.g. IEEE 754 number representation, UTF-16 code point sorting) typically requiring dedicated C libraries. In this architecture, PostgreSQL `jsonb` normalizes key order in memory, but wire-level determinism is achieved via **SQL-Level Deterministic Canonicalization**:
  1. Object keys are explicitly formatted in alphabetical order via `jsonb_build_object()`.
  2. Numbers are formatted to fixed explicit text strings (`ROUND(..., 6)::text` or integer paise strings).
  3. Timestamps are formatted as ISO 8601 UTC with microsecond precision: `to_char(ts AT TIME ZONE 'UTC', 'YYYY-MM-DD"T"HH24:MI:SS.US"Z"')`.
  4. Arrays are sorted by primary identity keys (`employee_id ASC`) using `jsonb_agg(... ORDER BY employee_id ASC)`.
  5. The resulting text is converted to bytea via `convert_to(v_canonical_text, 'UTF8')` and digested using `public.digest(..., 'sha256')`.
  - *Status:* **DESIGN-LEVEL SPECIFICATION ONLY (NOT CERTIFIED RFC 8785 IMPLEMENTATION)**.

### 3.3 Application GUC Context vs. Authenticated Caller Identity
- **Original Defect:** Identified in A3.13.1 §8. Prior text conflated application-provided GUCs with independent database-level user authentication.
- **Architectural Clarification:**

### 4.3 Durable Two-Phase Provisioning & Reconciliation Workflow
To handle distributed failures between Supabase GoTrue and PostgreSQL:

```
┌────────────────────────────────────────────────────────────────────────┐
│ STEP 1: Express Enqueues Provisioning Request                          │
│ INSERT INTO public.account_provisioning_requests (                     │

│ const { user, error } = await supabase.auth.admin.createUser({ ... }); │
│ IF error:                                                              │
│   UPDATE account_provisioning_requests SET status = 'FAILED';         │
```

```sql
  p_attendance_day_id UUID,

  p_net_payable_minutes INTEGER,
  p_approved_post18_minutes INTEGER,
  p_admin_user_id UUID,
  p_review_reason TEXT
)
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = pg_catalog, public
AS $$
DECLARE
  v_day RECORD;
  v_period_id UUID;
BEGIN
  -- 1. Read attendance day to resolve date
  SELECT id, organization_id, attendance_date, net_payable_minutes, day_status
  INTO v_day
  FROM public.attendance_days
  WHERE id = p_attendance_day_id;

    RAISE EXCEPTION 'Attendance day % not found', p_attendance_day_id USING ERRCODE = 'P0002';

  FROM public.payroll_periods
  WHERE organization_id = p_organization_id
    AND v_day.attendance_date BETWEEN start_date AND end_date
  FOR SHARE;

    RAISE EXCEPTION 'Attendance date % is not mapped to any configured payroll period', v_day.attendance_date
      USING ERRCODE = 'P0001';
  END IF;

      net_payable_minutes = p_net_payable_minutes,
      approved_post18_minutes = p_approved_post18_minutes,
      requires_admin_review = FALSE,
      reviewed_by = p_admin_user_id,
      reviewed_at = now(),
      review_reason = p_review_reason,
      updated_at = now()
  WHERE id = p_attendance_day_id;

  INSERT INTO public.audit_logs (
    organization_id, actor_user_id, actor_role, action, target_table, target_id,
    payload_before, payload_after, reason, created_at
```

### 5.3 Multi-Period Lock Ordering & Honest Concurrency Analysis
When an operation mutates records across multiple payroll periods (e.g., an offline biometric device syncing punches spanning October 31 and November 1):
1. **Sorted Lock Acquisition Rule:** The application/procedure must extract all distinct period IDs and sort them in **strict ascending UUID order**:
   ```sql
   SELECT id FROM public.payroll_periods WHERE id = ANY(v_sorted_period_ids) ORDER BY id ASC FOR SHARE;
   ```
2. **Deadlock Assessment:** While strict global ordering eliminates cycles among transactions locking *only* `payroll_periods`, **deadlocks cannot be claimed as mathematically eliminated** across the entire database. If concurrent transactions lock rows in `attendance_days` in differing order (e.g., Employee A day 1 vs day 2, while another transaction locks day 2 vs day 1), PostgreSQL's deadlock detector may abort a transaction.
3. **Engine-Level Bounded Defense:** Every transaction enforces `SET LOCAL lock_timeout = '3s';` and `SET LOCAL statement_timeout = '5s';`, triggering clean rollbacks and enabling the Express layer to retry with exponential backoff and jitter.

### 6.2 Hardened Express Transaction Wrapper
- **Original Defect:** Identified in A3.13.1 §6.2. Error suppression during rollback or release failure could mask underlying database exceptions.
- **Architectural Correction:** Provide the corrected TypeScript transaction wrapper with explicit error bubbling and socket destruction on failure:

```typescript
import { Pool, PoolClient } from 'pg';
import jwt, { JwtPayload } from 'jsonwebtoken';

const dbPool = new Pool({
  connectionString: process.env.DATABASE_URL,
  max: 50,
  idleTimeoutMillis: 30000,
});

export async function executeTenantTransaction<T>(
  rawJwtToken: string,
  callback: (client: PoolClient, context: { userId: string; orgId: string; role: string }) => Promise<T>
): Promise<T> {
  // 1. Verify JWT against configured Supabase signing algorithm and key
  const algorithm = (process.env.SUPABASE_JWT_ALGORITHM || 'HS256') as jwt.Algorithm;
  const secretOrKey = algorithm.startsWith('RS') || algorithm.startsWith('ES')
    ? process.env.SUPABASE_JWT_PUBLIC_KEY!
    : process.env.SUPABASE_JWT_SECRET!;

  const decoded = jwt.verify(rawJwtToken, secretOrKey, {
    algorithms: [algorithm],
    issuer: process.env.SUPABASE_JWT_ISSUER,
    audience: 'authenticated',
  }) as JwtPayload;

  const authUserId = decoded.sub;
  if (!authUserId) {
    throw new Error('AUTH_FAILED: Missing JWT sub claim');
  }

  // 2. Checkout dedicated client
  const client = await dbPool.connect();
  let isReleased = false;

  try {
    // 3. Begin transaction
    await client.query('BEGIN');

    // 4. Resolve authenticated tenant context directly from database
    const ctxRes = await client.query(
      'SELECT user_id, organization_id, primary_role, is_active, org_active FROM public.resolve_user_auth_context($1)',
      [authUserId]
    );

    const user = ctxRes.rows[0];
    if (!user || user.is_active === false || user.org_active === false) {
      throw new Error('AUTH_DENIED: User or organization account is deactivated');
    }

    // 5. Establish transaction-scoped GUCs
    await client.query(
      `SELECT
         set_config('app.current_user_id', $1, true),
         set_config('app.current_org_id', $2, true),
         set_config('app.current_user_role', $3, true)`,
      [user.user_id, user.organization_id, user.primary_role]
    );

    // 6. Execute business callback over verified client
    const result = await callback(client, {
      userId: user.user_id,
      orgId: user.organization_id,
      role: user.primary_role,
    });

    // 7. Commit transaction
    await client.query('COMMIT');
    return result;
  } catch (primaryError) {
    // 8. Rollback without suppressing primary business/DB error
    try {
      await client.query('ROLLBACK');
    } catch (rollbackError) {
      if (!isReleased) {
        isReleased = true;
        client.release(rollbackError); // Destroy broken socket
      }
      throw primaryError; // Always propagate original error
    }
    throw primaryError;
  } finally {
    // 9. Defensive cleanup ensuring socket is sanitized or destroyed
    if (!isReleased) {
      isReleased = true;
      try {
        await client.query('DISCARD SEQUENCES; RESET ALL;');
        client.release();
      } catch (cleanupError) {
        client.release(cleanupError); // Destroy contaminated socket
      }
    }
  }
}
```
