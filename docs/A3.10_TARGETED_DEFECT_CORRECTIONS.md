### 3.2 Complete Freeze Guard Trigger Definitions Across All Mutation Sources

Every table whose mutation could affect an active payroll compilation is protected by a dedicated `BEFORE` trigger. The trigger determines the affected working date range, queries `payroll_periods`, and immediately aborts the transaction if the period is frozen or closed:

```sql
-- 1. Freeze Guard Function for Attendance & Correction Entities
CREATE OR REPLACE FUNCTION trg_enforce_attendance_source_freeze_guard()
RETURNS TRIGGER AS $$
DECLARE
    v_working_date DATE;
    v_org_id UUID;
    v_period_frozen BOOLEAN;
    v_period_closed BOOLEAN;
    v_period_id UUID;
BEGIN
    -- Authoritative working date resolution
    IF TG_TABLE_NAME = 'attendance_days' THEN
        v_working_date := COALESCE(NEW.attendance_date, OLD.attendance_date);
        v_org_id := COALESCE(NEW.organization_id, OLD.organization_id);

    ELSIF TG_TABLE_NAME IN ('attendance_events', 'attendance_corrections', 'attendance_verifications') THEN
        -- Resolve working date directly from parent attendance_days
        SELECT ad.attendance_date, ad.organization_id INTO v_working_date, v_org_id
        FROM attendance_days ad
        WHERE ad.id = COALESCE(NEW.attendance_day_id, OLD.attendance_day_id);

        IF NOT FOUND THEN
            RAISE EXCEPTION 'Integrity violation: Parent attendance_day % does not exist (SQLSTATE 23503)', 
                COALESCE(NEW.attendance_day_id, OLD.attendance_day_id)
                USING ERRCODE = 'foreign_key_violation';
        END IF;
    END IF;

    -- Lookup period status under shared lock
    SELECT id, is_frozen, is_closed INTO v_period_id, v_period_frozen, v_period_closed
    FROM payroll_periods
    WHERE organization_id = v_org_id
      AND v_working_date BETWEEN start_date AND end_date
    FOR SHARE;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Integrity violation: Working date % is not mapped to any configured payroll period (SQLSTATE P0001)', v_working_date
            USING ERRCODE = 'check_violation';
    END IF;

    IF v_period_frozen = TRUE THEN
        RAISE EXCEPTION 'Freeze violation: Period % is FROZEN pending approval. Mutation on % rejected (SQLSTATE P0002)', v_period_id, TG_TABLE_NAME
            USING ERRCODE = 'check_violation';
    END IF;

    IF v_period_closed = TRUE THEN
        RAISE EXCEPTION 'Freeze violation: Period % is PERMANENTLY CLOSED. Mutation on % rejected (SQLSTATE P0003)', v_period_id, TG_TABLE_NAME
            USING ERRCODE = 'check_violation';
    END IF;

    RETURN COALESCE(NEW, OLD);
END;
$$ LANGUAGE plpgsql SET search_path = pg_catalog, public;

-- Attach trigger to attendance sources
CREATE TRIGGER trg_freeze_guard_attendance_days
BEFORE INSERT OR UPDATE OR DELETE ON attendance_days
FOR EACH ROW EXECUTE FUNCTION trg_enforce_attendance_source_freeze_guard();

CREATE TRIGGER trg_freeze_guard_attendance_events
BEFORE INSERT ON attendance_events
FOR EACH ROW EXECUTE FUNCTION trg_enforce_attendance_source_freeze_guard();

CREATE TRIGGER trg_freeze_guard_attendance_corrections
BEFORE INSERT ON attendance_corrections
FOR EACH ROW EXECUTE FUNCTION trg_enforce_attendance_source_freeze_guard();

CREATE TRIGGER trg_freeze_guard_attendance_verifications
BEFORE INSERT ON attendance_verifications
FOR EACH ROW EXECUTE FUNCTION trg_enforce_attendance_source_freeze_guard();

-- 2. Freeze Guard Function for Compensation & Calendar Sources (Range-Based)
CREATE OR REPLACE FUNCTION trg_enforce_range_source_freeze_guard()
RETURNS TRIGGER AS $$
DECLARE
    v_org_id UUID;
    v_start DATE;
    v_end DATE;
    v_frozen_count INTEGER;
BEGIN
    IF TG_TABLE_NAME = 'employee_salaries' THEN
        v_org_id := COALESCE(NEW.organization_id, OLD.organization_id);
        v_start := COALESCE(NEW.effective_from, OLD.effective_from);
        v_end := COALESCE(NEW.effective_to, OLD.effective_to, 'infinity'::date);

    ELSIF TG_TABLE_NAME = 'leave_applications' THEN
        v_org_id := COALESCE(NEW.organization_id, OLD.organization_id);
        v_start := COALESCE(NEW.start_date, OLD.start_date);
        v_end := COALESCE(NEW.end_date, OLD.end_date);

    ELSIF TG_TABLE_NAME = 'holidays' THEN
        v_org_id := COALESCE(NEW.organization_id, OLD.organization_id);
        v_start := COALESCE(NEW.holiday_date, OLD.holiday_date);
        v_end := v_start;

    ELSIF TG_TABLE_NAME = 'employee_payroll_adjustments' THEN
        SELECT pp.organization_id, pp.start_date, pp.end_date INTO v_org_id, v_start, v_end
        FROM payroll_periods pp WHERE pp.id = COALESCE(NEW.payroll_period_id, OLD.payroll_period_id);
    END IF;

    -- Check if ANY overlapping period is currently frozen
    SELECT COUNT(*) INTO v_frozen_count
    FROM payroll_periods
    WHERE organization_id = v_org_id
      AND (start_date <= v_end AND end_date >= v_start)
      AND is_frozen = TRUE;

    IF v_frozen_count > 0 THEN
        RAISE EXCEPTION 'Freeze violation: Operation overlaps with a FROZEN payroll period on table % (SQLSTATE P0002)', TG_TABLE_NAME
            USING ERRCODE = 'check_violation';
    END IF;

    RETURN COALESCE(NEW, OLD);
END;
$$ LANGUAGE plpgsql SET search_path = pg_catalog, public;

CREATE TRIGGER trg_freeze_guard_employee_salaries
BEFORE INSERT OR UPDATE OR DELETE ON employee_salaries
FOR EACH ROW EXECUTE FUNCTION trg_enforce_range_source_freeze_guard();

CREATE TRIGGER trg_freeze_guard_leave_applications
BEFORE INSERT OR UPDATE OR DELETE ON leave_applications
FOR EACH ROW EXECUTE FUNCTION trg_enforce_range_source_freeze_guard();

CREATE TRIGGER trg_freeze_guard_holidays
BEFORE INSERT OR UPDATE OR DELETE ON holidays
FOR EACH ROW EXECUTE FUNCTION trg_enforce_range_source_freeze_guard();

CREATE TRIGGER trg_freeze_guard_payroll_adjustments
BEFORE INSERT OR UPDATE OR DELETE ON employee_payroll_adjustments
FOR EACH ROW EXECUTE FUNCTION trg_enforce_range_source_freeze_guard();
```

### 3.3 Canonical RFC 8785 JSONB Snapshot Serialization & Cryptographic Staleness

To eliminate delimiter injection vulnerabilities, serialization utilizes **Canonical JSONB**. In PostgreSQL, `jsonb` natively guarantees sorted key ordering, strict UTF-8 escaping, whitespace normalization, and unambiguous null representations.

#### The Authoritative Staleness Detection Invariant:
We eliminate reliance on `SUM(version)`. Staleness is detected with mathematical certainty via the **Cryptographic SHA-256 Digest**:
$$\text{IsStale} = (\text{CompiledSnapshotHash} \ne \text{LiveSnapshotHash})$$
Because SHA-256 is collision-resistant ($2^{-256}$ probability of accidental collision), any insertion, update, or deletion across any of the 11 source tables deterministically alters the canonical digest.

```sql
CREATE OR REPLACE FUNCTION generate_payroll_source_snapshot(
    p_payroll_period_id UUID,
    p_organization_id UUID
)
RETURNS TABLE (
    snapshot_hash VARCHAR(64),
    canonical_payload JSONB
) AS $$
DECLARE
    v_start_date DATE;
    v_end_date DATE;
    v_json JSONB;
BEGIN
    -- 1. Retrieve period date boundaries under transaction lock
    SELECT start_date, end_date INTO v_start_date, v_end_date
    FROM payroll_periods
    WHERE id = p_payroll_period_id AND organization_id = p_organization_id;

    -- 2. Build canonical, deterministically sorted JSONB across all source tables
    SELECT jsonb_build_object(
        'schema_version', 'HRMS_PAYROLL_SNAPSHOT_v1.0',
        'period_id', p_payroll_period_id,
        'organization_id', p_organization_id,
        'start_date', v_start_date,
        'end_date', v_end_date,
        
        -- Working Days with inner aggregates
        'days', COALESCE((
            SELECT jsonb_agg(
                jsonb_build_object(
                    'day_id', ad.id,
                    'employee_id', ad.employee_id,
                    'date', ad.attendance_date,
                    'status', ad.day_status,
                    'payable_minutes', ad.net_payable_minutes,
                    'version', ad.version,
                    'updated_at', to_char(ad.updated_at AT TIME ZONE 'UTC', 'YYYY-MM-DD"T"HH24:MI:SS.US"Z"'),
                    'events', COALESCE((
                        SELECT jsonb_agg(
                            jsonb_build_object(
                                'event_id', ae.id,
                                'type', ae.event_type,
                                'timestamp', to_char(ae.server_timestamp AT TIME ZONE 'UTC', 'YYYY-MM-DD"T"HH24:MI:SS.US"Z"')
                            ) ORDER BY ae.id
                        ) FROM attendance_events ae WHERE ae.attendance_day_id = ad.id
                    ), '[]'::jsonb),
                    'corrections', COALESCE((
                        SELECT jsonb_agg(
                            jsonb_build_object(
                                'correction_id', ac.id,
                                'action', ac.action_type,
                                'corrected_time', to_char(ac.corrected_timestamp AT TIME ZONE 'UTC', 'YYYY-MM-DD"T"HH24:MI:SS.US"Z"'),
                                'created_by', ac.created_by
                            ) ORDER BY ac.id
                        ) FROM attendance_corrections ac WHERE ac.attendance_day_id = ad.id
                    ), '[]'::jsonb),
                    'verifications', COALESCE((
                        SELECT jsonb_agg(
                            jsonb_build_object(
                                'verification_id', av.id,
                                'status', av.verification_status,
                                'reviewer', av.reviewed_by,
                                'reviewed_at', to_char(av.created_at AT TIME ZONE 'UTC', 'YYYY-MM-DD"T"HH24:MI:SS.US"Z"')
                            ) ORDER BY av.id
                        ) FROM attendance_verifications av WHERE av.attendance_day_id = ad.id
                    ), '[]'::jsonb)
                ) ORDER BY ad.id
            ) FROM attendance_days ad
            WHERE ad.organization_id = p_organization_id
              AND ad.attendance_date BETWEEN v_start_date AND v_end_date
        ), '[]'::jsonb),

        -- Employee Compensation Contracts
        'salaries', COALESCE((
            SELECT jsonb_agg(
                jsonb_build_object(
                    'salary_id', es.id,
                    'employee_id', es.employee_id,
                    'paise', es.monthly_salary_reference_paise,
                    'from', es.effective_from,
                    'to', es.effective_to
                ) ORDER BY es.id
            ) FROM employee_salaries es
            WHERE es.organization_id = p_organization_id
              AND es.effective_from <= v_end_date
              AND (es.effective_to IS NULL OR es.effective_to >= v_start_date)
        ), '[]'::jsonb),

        -- Mid-month Employment Proration Inputs
        'employees', COALESCE((
            SELECT jsonb_agg(
                jsonb_build_object(
                    'employee_id', e.id,
                    'joining_date', e.joining_date,
                    'status', e.employment_status
                ) ORDER BY e.id
            ) FROM employees e
            WHERE e.organization_id = p_organization_id
              AND e.joining_date <= v_end_date
        ), '[]'::jsonb),

        -- Approved Payroll Adjustments
        'adjustments', COALESCE((
            SELECT jsonb_agg(
                jsonb_build_object(
                    'adjustment_id', epa.id,
                    'employee_id', epa.employee_id,
                    'type', epa.adjustment_type,
                    'amount_paise', epa.amount_paise
                ) ORDER BY epa.id
            ) FROM employee_payroll_adjustments epa
            WHERE epa.organization_id = p_organization_id
              AND epa.payroll_period_id = p_payroll_period_id
              AND epa.status = 'PENDING_PAYROLL'
        ), '[]'::jsonb),

        -- Approved Leaves
        'leaves', COALESCE((
            SELECT jsonb_agg(
                jsonb_build_object(
                    'leave_id', la.id,
                    'employee_id', la.employee_id,
                    'is_paid', lt.is_paid,
                    'start_date', la.start_date,
                    'end_date', la.end_date
                ) ORDER BY la.id
            ) FROM leave_applications la
            JOIN leave_types lt ON la.leave_type_id = lt.id
            WHERE la.organization_id = p_organization_id
              AND la.status = 'APPROVED'
              AND (la.start_date <= v_end_date AND la.end_date >= v_start_date)
        ), '[]'::jsonb),

        -- Holidays
        'holidays', COALESCE((
            SELECT jsonb_agg(
                jsonb_build_object(
                    'holiday_id', h.id,
                    'date', h.holiday_date
                ) ORDER BY h.id
            ) FROM holidays h
            WHERE h.organization_id = p_organization_id
              AND h.holiday_date BETWEEN v_start_date AND v_end_date
        ), '[]'::jsonb)
    ) INTO v_json;

    -- Compute SHA-256 Digest over raw canonical UTF-8 JSON text
    snapshot_hash := encode(public.digest(convert_to(v_json::text, 'UTF8'), 'sha256'), 'hex');
    canonical_payload := v_json;
    RETURN NEXT;
END;
$$ LANGUAGE plpgsql STABLE SET search_path = pg_catalog, public;
```

### 4.1 Multi-Parent Consistency Triggers

Where child entities reference multiple parents, the database enforces cross-parent consistency via triggers, guaranteeing both parents belong to the same tenant:

```sql
-- 1. Multi-Parent Trigger for payroll_items (Verifies employee belongs to same tenant as payroll run)
CREATE OR REPLACE FUNCTION trg_validate_payroll_item_tenant_consistency()
RETURNS TRIGGER AS $$
DECLARE
    v_run_org_id UUID;
    v_emp_org_id UUID;
BEGIN
    SELECT pp.organization_id INTO v_run_org_id
    FROM payroll_runs pr
    JOIN payroll_periods pp ON pr.payroll_period_id = pp.id
    WHERE pr.id = NEW.payroll_run_id;

    SELECT organization_id INTO v_emp_org_id
    FROM employees WHERE id = NEW.employee_id;

    IF v_run_org_id IS DISTINCT FROM v_emp_org_id THEN
        RAISE EXCEPTION 'Cross-tenant violation: Employee % (org %) does not match payroll run % (org %)',
            NEW.employee_id, v_emp_org_id, NEW.payroll_run_id, v_run_org_id
            USING ERRCODE = 'check_violation';
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql SET search_path = pg_catalog, public;

CREATE TRIGGER trg_check_payroll_items_tenant_consistency
BEFORE INSERT OR UPDATE ON payroll_items
FOR EACH ROW EXECUTE FUNCTION trg_validate_payroll_item_tenant_consistency();

-- 2. Multi-Parent Trigger for attendance_events (Verifies attendance_day_id belongs to same employee)
CREATE OR REPLACE FUNCTION trg_validate_attendance_event_employee_match()
RETURNS TRIGGER AS $$
DECLARE
    v_day_emp_id UUID;
BEGIN
    SELECT employee_id INTO v_day_emp_id
    FROM attendance_days WHERE id = NEW.attendance_day_id;

            NEW.employee_id, v_day_emp_id
            USING ERRCODE = 'check_violation';
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql SET search_path = pg_catalog, public;

BEFORE INSERT OR UPDATE ON attendance_events
FOR EACH ROW EXECUTE FUNCTION trg_validate_attendance_event_employee_match();
```

## 6. Finding D — Globally Consistent Lock Ordering Hierarchy

### 6.1 Top-Down 4-Level Lock Hierarchy

To eliminate **DEF-10.4** (lock hierarchy inversion), every transaction modifying attendance or payroll data MUST acquire locks in the exact top-down relational hierarchy:

```
Level 1: payroll_periods (Parent Root Lock)
   ↓
Level 2: payroll_runs (Batch Run Lock)
   ↓
Level 3: attendance_days (Working Day Aggregate Lock)
   ↓
Level 4: Child Sink Tables (attendance_events, attendance_corrections, attendance_verifications, payroll_items)
```

#### The Elimination of Deadlock Inversion:
- **Prior Flawed Sequence:** Attendance writer locked `attendance_days` (Level 3) $\rightarrow$ trigger locked `payroll_periods` (Level 1). Compiler locked `payroll_periods` (Level 1) $\rightarrow$ calculated over `attendance_days` (Level 3). $\implies$ **Deadlock!**
- **Corrected Top-Down Sequence:**
  1. Attendance writer issues `SELECT id, is_frozen, is_closed FROM payroll_periods WHERE ... FOR SHARE;` (Level 1 Lock Acquired First).
  2. Attendance writer evaluates `is_frozen = FALSE` and `is_closed = FALSE`.
  3. Attendance writer issues `SELECT ... FROM attendance_days WHERE ... FOR UPDATE;` (Level 3 Lock Acquired Second).
  4. Attendance writer inserts into `attendance_events` (Level 4).
- *Mathematical Proof:* Because **both** compiler and attendance writers acquire Level 1 locks before touching Level 3 or 4 rows, lock acquisition order is strictly monotonic ($L_1 \prec L_2 \prec L_3 \prec L_4$). Lock hierarchy inversion is **mathematically impossible**!

---

### 6.2 Multi-Period Lock Ordering & Deadlock Freedom

When offline punch synchronization uploads attendance batches spanning multiple period boundaries:
1. Express extracts all distinct `payroll_period_id` values.
2. Express issues the locking query with deterministic ascending sorting:
   ```sql
   SELECT id, is_frozen, is_closed
   FROM payroll_periods
   WHERE id = ANY(:distinct_period_ids)
   ORDER BY id ASC
   FOR SHARE;
   ```
3. Because all worker threads lock multiple periods in the identical ascending UUID order, the resource allocation graph contains zero directed cycles. Distributed deadlocks (`SQLSTATE 40P01`) are eliminated.

```typescript
// Express Gateway Complete Request Lifecycle Pattern
export async function executeAuthenticatedTransaction<T>(
    token: string,
    operation: (client: PoolClient, userContext: UserContext) => Promise<T>
): Promise<T> {
    // Step 1: Local JWT Verification (Signature, Expiration, Issuer)
    const decoded = jwt.verify(token, process.env.SUPABASE_JWT_SECRET!, {
        issuer: process.env.SUPABASE_JWT_ISSUER,
        audience: 'authenticated'
    }) as JwtPayload;

    const client = await dbPool.connect();
    try {
        await client.query('BEGIN');

        // Step 2: Query active status via RLS-exempt auth context function
        const authRes = await client.query(
            'SELECT user_id, organization_id, is_active FROM resolve_user_auth_context($1)',
            [decoded.sub]
        );
        const user = authRes.rows[0];

        if (!user || user.is_active === false) {
            throw new AuthenticationError('ACCOUNT_DEACTIVATED');
        }

        // Step 3: Establish Transaction-Local RLS and Actor Context (is_local = true)
        await client.query(
            "SELECT set_config('app.current_org_id', $1, true), set_config('app.current_user_id', $2, true)",
            [user.organization_id, user.user_id]
        );

        // Step 4: Execute Business Operation
        const result = await operation(client, user);

        await client.query('COMMIT');
        return result;
    } catch (err) {
        await client.query('ROLLBACK');
        throw err;
    } finally {
        // Pool client is released back to pool clean (GUCs cleared by ROLLBACK/COMMIT)
        client.release();
    }
}
```

*Guarantees:*
- `is_local = true` in `set_config` guarantees that `app.current_org_id` and `app.current_user_id` exist strictly for the lifespan of the transaction and are automatically destroyed upon `COMMIT` or `ROLLBACK`.

```python
def execute_topological_sort(nodes: list[str], edges: list[tuple[str, str]]) -> list[str]:
    from collections import defaultdict, deque
    
    in_degree = {n: 0 for n in nodes}
    adj = defaultdict(list)
    
    for u, v in edges:
        adj[u].append(v)
        in_degree[v] += 1
        
    queue = deque([n for n in nodes if in_degree[n] == 0])
    ordered = []
    
    while queue:
        curr = queue.popleft()
        ordered.append(curr)
        for neighbor in adj[curr]:
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)
                
    if len(ordered) != len(nodes):
        unresolved = [n for n, deg in in_degree.items() if deg > 0]
        raise ValueError(f"Cycle detected! Unresolved objects: {unresolved}")
        
    return ordered
```
